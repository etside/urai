function updateSendBtn() {
  var v = $('user-input').value;
  $('send-btn').disabled = (!v.trim() && !attachedImg) || !!abortCtrl;
  $('char-count').textContent = v.length > 50 ? v.length + ' chars' : '';

function handleKey(e) {

}

/* ── Image attach / voice ───────────────────────────────────────────────── */
function attachImage(e) {
  var file = e.target.files[0];
  if (!file) return;
  var reader = new FileReader();
  reader.onload = function (ev) {
    attachedImg = ev.target.result;
    var preview = $('img-preview');
    preview.src = attachedImg;
    preview.style.display = 'block';
    updateSendBtn();
  };
  reader.readAsDataURL(file);
}

function startVoice() {
  if (!('webkitSpeechRecognition' in window || 'SpeechRecognition' in window)) {
    alert('Voice input not supported in this browser'); return;
  }
  var SR = window.SpeechRecognition || window.webkitSpeechRecognition;
  recognition = new SR();
  recognition.lang = 'en-US';
  recognition.interimResults = true;
  var btn = $('voice-btn');
  btn.textContent = '🔴';
  recognition.onresult = function (e) {
    $('user-input').value = e.results[e.results.length - 1][0].transcript;
    updateSendBtn();
  };
  recognition.onend = function () { btn.textContent = '🎤'; };
  recognition.start();
}

async function checkTokenBanner() {
  var s = await api().tokenStatus();
  var b = $('token-banner');
  if (!s) return;
  if (s.blocked_until > 0) {
    b.classList.add('is-open');
    b.innerHTML = 'Free Token Expired / 0 available Token. Submit master key 856777 to restore access. ';
    openTokenModal();
  } else if (!api().getKey() && s.remaining <= 0) {
    b.classList.add('is-open');
    b.textContent = 'Free Token Expired / 0 available Token. Submit master key 856777.';
  } else {
    b.classList.remove('is-open');
  }
  var rk = $('req-dev');
  if (rk && !rk.value) rk.value = api().deviceId();
}

function openTokenModal() {
  $('token-overlay').classList.add('is-open');
  setTimeout(function () { var i = $('master-key-input'); if (i) i.focus(); }, 50);
}
function closeTokenModal() { $('token-overlay').classList.remove('is-open'); }

async function submitMasterKey() {
  var k = $('master-key-input').value.trim();
  var m = $('token-msg');
  try {
    var d = await api().unlock(k);
    if (d && d.ok) {
      m.style.color = 'var(--accent)';
      m.textContent = 'Unlocked. Cost 0.';
      closeTokenModal(); checkTokenBanner(); refreshTokenMeter();
      toast('Master key accepted — unlimited, cost 0', 'ok');
    } else {
      m.style.color = 'var(--danger)';
      m.textContent = (d && d.message) || 'Invalid key';
    }
  } catch (e) {
    m.style.color = 'var(--danger)';
    m.textContent = errText(e, 'Unlock failed');
  }
}

async function agentPlanRequest(text) {

async function studioGenerate() {
  if (studioBusy) return;
  var M = window.UraiMedia;
  if (!M) { toast('Media client not loaded', 'err'); return; }
  var prompt = ($('st-vid-prompt') || {}).value ? $('st-vid-prompt').value.trim() : '';
  var profileId = ($('st-vid-profile') || {}).value || 'ref2va_vsa';
  var meta = M.profileMeta(profileId) || { kind: 'video', provider: 'auto' };
  if (!prompt) { toast('Enter a prompt', 'err'); return; }

  var msg = $('st-vid-msg'); if (msg) msg.innerHTML = '';
  studioBusy = true;
  setStudioBusy(true);
  var state = $('st-vid-state');
  if (state) state.innerHTML = '
Generating… ref2va warms in ~72s on an RTX 4090; the local animatic is faster but less realistic.
';
  try {
    if (meta.kind === 'image') {
      var img = await M.generateImage({ prompt: prompt });
      if (img.ok) {
        addStudioMedia(img.url, 'image');
        recordStudioJob(Object.assign({ provider: 'seedream4', provider_label: 'Seedream 4.0 (image)' }, img.raw || {}),
          { kind: 'image', prompt: prompt, profile: profileId });
        toast('Image ready', 'ok');
      } else {
        studioRenderError('Image generation returned no media.');
      }
      return;
    }
    var duration = parseInt(($('st-vid-duration') || {}).value, 10) || 5;
    var res = await M.generateVideo({
      prompt: prompt,
      profile: profileId,
      provider: M.providerFor(profileId),
      duration: duration,
      ref_images: studioRefs.slice(),
      response_format: 'b64_json'
    });
    if (res.installed === false || (!res.ok && M.isNotInstalled(res))) {
      await studioRenderNotInstalled(res, profileId);
    } else if (!res.ok) {
      studioRenderError(res.error || 'Generation failed.');
    } else {
      addStudioJob(res, { prompt: prompt, kind: 'video', profile: profileId });
      toast('Video ready', 'ok');
    }
  } catch (e) {
    studioRenderError(errText(e));
  } finally {
    studioBusy = false;
    setStudioBusy(false);
    var st = $('st-vid-state'); if (st) st.innerHTML = '';
  }
}

function initMotion() {
  try {
    if (window.matchMedia('(prefers-reduced-motion: reduce)').matches) return;
    var cv = $('bg-fx');
    if (cv) {
      var ctx = cv.getContext('2d');
      var w, h, t = 0, running = true;
      var DPR = Math.min(window.devicePixelRatio || 1, 1.5);
      var blobs = [
        { x: 0.2, y: 0.25, r: 0.34, c: '0,96,252', sp: 0.00021, ph: 0 },
        { x: 0.85, y: 0.7, r: 0.38, c: '18,176,240', sp: 0.00016, ph: 2 },
        { x: 0.6, y: 0.1, r: 0.26, c: '0,216,144', sp: 0.00026, ph: 4 }
      ];
      function size() {
        w = cv.width = innerWidth * DPR; h = cv.height = innerHeight * DPR;
        cv.style.width = innerWidth + 'px'; cv.style.height = innerHeight + 'px';
      }
      size(); addEventListener('resize', size);
      document.addEventListener('visibilitychange', function () {
        running = !document.hidden; if (running) requestAnimationFrame(frame);
      });
      function frame() {
        if (!running) return;
        t++;
        ctx.clearRect(0, 0, w, h);
        ctx.globalCompositeOperation = 'lighter';
        for (var i = 0; i < blobs.length; i++) {
          var b = blobs[i];
          var bx = (b.x + Math.sin(t * b.sp + b.ph) * 0.06) * w;
          var by = (b.y + Math.cos(t * b.sp * 1.3 + b.ph) * 0.06) * h;
          var br = b.r * Math.min(w, h);
          var g = ctx.createRadialGradient(bx, by, 0, bx, by, br);
          g.addColorStop(0, 'rgba(' + b.c + ',.10)');
          g.addColorStop(1, 'rgba(' + b.c + ',0)');
          ctx.fillStyle = g;
          ctx.fillRect(0, 0, w, h);
        }
        requestAnimationFrame(frame);
      }
      requestAnimationFrame(frame);
    }
    document.addEventListener('pointermove', function (e) {
      var el = e.target.closest ? e.target.closest('.sug-btn, .tilt') : null;
      if (!el) return;
      var r = el.getBoundingClientRect();
      var px = (e.clientX - r.left) / r.width - 0.5;
      var py = (e.clientY - r.top) / r.height - 0.5;
      el.style.setProperty('--ry', (px * 14).toFixed(2) + 'deg');
      el.style.setProperty('--rx', (-py * 14).toFixed(2) + 'deg');
    });
    document.addEventListener('pointerout', function (e) {
      var el = e.target.closest ? e.target.closest('.sug-btn, .tilt') : null;
      if (el && !(e.relatedTarget && e.relatedTarget.closest && e.relatedTarget.closest('.sug-btn, .tilt') === el)) {
        el.style.setProperty('--rx', '0deg'); el.style.setProperty('--ry', '0deg');
      }
    });
  } catch (e) { /* motion is decorative — never break chat */ }
}
